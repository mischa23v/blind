cmake_minimum_required(VERSION 3.16)
project(KoivuRtabComponent LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(RTABMap REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs)
find_package(PCL REQUIRED COMPONENTS common io)
add_executable(koivu_rtab_component main.cpp)
target_include_directories(koivu_rtab_component PRIVATE ${RTABMap_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
target_link_libraries(koivu_rtab_component PRIVATE ${RTABMap_LIBRARIES} ${OpenCV_LIBRARIES} ${PCL_LIBRARIES})
add_executable(koivu_rtab_mapper mapper.cpp)
target_include_directories(koivu_rtab_mapper PRIVATE ${RTABMap_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
target_link_libraries(koivu_rtab_mapper PRIVATE ${RTABMap_LIBRARIES} ${OpenCV_LIBRARIES} ${PCL_LIBRARIES})
add_executable(koivu_rtab_export export.cpp)
target_include_directories(koivu_rtab_export PRIVATE ${RTABMap_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
target_link_libraries(koivu_rtab_export PRIVATE ${RTABMap_LIBRARIES} ${OpenCV_LIBRARIES} ${PCL_LIBRARIES})

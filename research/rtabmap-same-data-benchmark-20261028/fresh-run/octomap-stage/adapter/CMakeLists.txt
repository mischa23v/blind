cmake_minimum_required(VERSION 3.16)
project(KoivuNativeOccupancy LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(RTABMap REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
add_executable(koivu_native_occupancy main.cpp)
target_include_directories(koivu_native_occupancy PRIVATE ${RTABMap_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
target_link_libraries(koivu_native_occupancy PRIVATE ${RTABMap_LIBRARIES} ${PCL_LIBRARIES})
